cmake_minimum_required(VERSION 3.20)
project(halo_ink_pen LANGUAGES CXX)
find_package(JNI REQUIRED)
add_library(halo_ink_pen SHARED pen_input.cpp)
target_compile_features(halo_ink_pen PRIVATE cxx_std_17)
target_include_directories(halo_ink_pen PRIVATE ${JNI_INCLUDE_DIRS})
if(WIN32)
    target_compile_definitions(halo_ink_pen PRIVATE _WIN32_WINNT=0x0602 UNICODE _UNICODE)
    target_link_libraries(halo_ink_pen PRIVATE user32)
endif()
